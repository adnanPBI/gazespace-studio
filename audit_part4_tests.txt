class Node:
    def __init__(self, value, *, transform=None, reject=False):
        self._value, self.transform, self.reject = value, transform, reject

    @property
    def Value(self):
        return self._value

    @Value.setter
    def Value(self, value):
        if self.reject:
            raise ValueError("node is not writable in this camera mode")
        self._value = self.transform(value) if self.transform else value


def fake_pylon(monkeypatch, **nodes):
    class Camera:
        def Open(self): self.open = True
        def IsOpen(self): return self.open
        def Close(self): self.open = False
        def StartGrabbing(self, *_): self.grabbing = True
        def IsGrabbing(self): return self.grabbing
        def StopGrabbing(self): self.grabbing = False
    camera = Camera()
    for name, node in nodes.items():
        setattr(camera, name, node)
    device = SimpleNamespace(GetSerialNumber=lambda: "000123", GetModelName=lambda: "control fixture")
    factory = SimpleNamespace(EnumerateDevices=lambda: [device], CreateDevice=lambda d: d)
    pylon = SimpleNamespace(TlFactory=SimpleNamespace(GetInstance=lambda: factory),
                            InstantCamera=lambda d: camera,
                            ImageFormatConverter=lambda: SimpleNamespace(),
                            PixelType_BGR8packed=3, GrabStrategy_LatestImageOnly=1)
    monkeypatch.setitem(sys.modules, "pypylon", SimpleNamespace(pylon=pylon))
    return camera


def test_basler_factory_applies_resolution_and_legacy_rate_without_enable_node(monkeypatch):
    camera = fake_pylon(monkeypatch, Width=Node(1024), Height=Node(768),
                        AcquisitionFrameRateAbs=Node(60.))
    with create_source("000123", backend="pylon", width=640, height=480, fps=25.) as source:
        report = source.capabilities()
        assert (report["width"], report["height"], report["fps_reported"]) == (640, 480, 25.)
        assert report["serial"] == "000123"
        for name in ("Width", "Height", "AcquisitionFrameRateAbs"):
            assert report["controls"][name]["applied"]
            assert report["controls"][name]["readback_matches"]
        assert "AcquisitionFrameRateEnable" not in report["controls"]
    assert not camera.open

from copy import deepcopy

def observation(stamp, *, features=None, quality=.9, blink=False, mode='ir-roi'):
    return Observation(stamp, (640, 480), features=np.asarray(features if features is not None else [.5] * 10),
                       quality=quality, blink=blink, eyes_open=not blink,
                       pupil_method='ir_manual_roi', context={'tracking_mode':mode})


def test_counts_unique_frames_and_requires_real_dwell_duration():
    collector = TargetCollector('center', (.5, .5), min_samples=4, min_duration_s=.6)
    collector.start(0)
    for t in np.arange(0, .6, .05):
        assert not collector.add(observation(float(t)))
    assert collector.add(observation(.6))
    for _ in range(20):
        assert not collector.add(observation(.6))
    for t in (.65, .7, .75):
        collector.add(observation(t))
    assert len(collector.observations) == 4
    assert not collector.complete
    for t in (.8, .9, 1., 1.1, 1.2):
        collector.add(observation(t))
    assert collector.complete
    assert collector.progress()['fraction'] == 1
    assert collector.progress()['stable_duration_s'] >= .59


def test_blink_discards_partial_target_and_restabilizes():
    collector = TargetCollector('center', (.5, .5), min_samples=4, min_duration_s=.3)
    collector.start(0)

    assert collector.observations
    assert not collector.add(observation(.85, blink=True))

    assert collector.reason == 'blink_or_closed_eyes'

        collector.add(observation(float(t)))
    assert collector.complete
    assert collector.observations[0].timestamp_s >= 1.45 - .001

def generate_samples(outliers=False):
    rng=np.random.default_rng(7)
    records=[]
    for i,(x,y) in enumerate((x,y) for x in np.linspace(.08,.92,4) for y in np.linspace(.08,.92,4)):
        for k in range(14):
            features=np.array([x,y,x,y,0,0,0,.5,.5,.72])+np.r_[rng.normal(0,.006,4),np.zeros(6)]
            target=[x,y]
            if outliers and k==0: target=[1-x,1-y]
            records.append({"features":features.tolist(),"target":target,"target_id":str(i)})
    return records

def checkerboard_view(index=0,width=1200,height=800):
    # Warped raster test: validates board corner extraction but does not represent
    # independently measured physical camera geometry.
    sq=55
    board=np.full((7*sq+120,10*sq+120,3),255,np.uint8)
    for row in range(7):
        for col in range(10):
            if (row+col)%2==0:
                cv2.rectangle(board,(60+col*sq,60+row*sq),
                              (60+(col+1)*sq,60+(row+1)*sq),(0,0,0),-1)
    src=np.float32([[0,0],[board.shape[1]-1,0],[board.shape[1]-1,board.shape[0]-1],[0,board.shape[0]-1]])
    i=index

"""Windows clock regressions: coarse legacy ticks and different clock epochs.

Linux commonly backs monotonic/perf_counter with the same clock. Deliberately
separating them exposes mixed-domain freshness bugs without needing a camera.
"""
from __future__ import annotations

import sys
from threading import Event
from types import SimpleNamespace

import cv2
import numpy as np
import pytest

import gazespace.clock as clock_module
from gazespace.capture import FrameSource, OpenCVSource, PylonSource, SyntheticEyeSource
from gazespace.clock import HOST_CLOCK_DOMAIN, host_now_s
from gazespace.engine import GazeEngine
from gazespace.runtime import TrackingRuntime
from gazespace.sessions import SessionRecorder, load_session
from gazespace.types import Frame, GazeResult, Observation


class HostClock:
    def __init__(self, value=100_000.0, step=0.0):
        self.value, self.step = value, step

    def __call__(self):
        current = self.value
        self.value += self.step
        return current


def separate_clocks(monkeypatch, *, step=0.0):
    host = HostClock(step=step)
    # Many rapid reads share a legacy Windows tick, at a different epoch.
    monkeypatch.setattr(clock_module.time, "monotonic", lambda: 17.0)
    monkeypatch.setattr(clock_module.time, "perf_counter", host)
    return host


def fake_pylon(monkeypatch):
    class Node:
        def __init__(self, value): self.Value = value

    class Result:
        PixelType, TimeStamp = 1, 987654321
        Array = np.full((12, 16), 127, np.uint8)
        def GrabSucceeded(self): return True
        def Release(self): pass

    class Camera:
        Width, Height = Node(16), Node(12)
        def Open(self): self.open = True
        def IsOpen(self): return self.open
        def Close(self): self.open = False
        def StartGrabbing(self, *_): self.grabbing = True
        def IsGrabbing(self): return self.grabbing
        def StopGrabbing(self): self.grabbing = False
        def RetrieveResult(self, *_): return Result()

    device = SimpleNamespace(GetSerialNumber=lambda: "00017", GetModelName=lambda: "Clock fixture")
    factory = SimpleNamespace(EnumerateDevices=lambda: [device], CreateDevice=lambda value: value)
    pylon = SimpleNamespace(TlFactory=SimpleNamespace(GetInstance=lambda: factory),
                            InstantCamera=lambda _: Camera(), ImageFormatConverter=lambda: SimpleNamespace(),
                            PixelType_Mono8=1, PixelType_BGR8packed=3,
                            GrabStrategy_LatestImageOnly=1, TimeoutHandling_ThrowException=1)
    monkeypatch.setitem(sys.modules, "pypylon", SimpleNamespace(pylon=pylon))


@pytest.mark.parametrize("backend", ["opencv", "pylon"])
def test_rapid_live_capture_uses_high_resolution_host_clock(monkeypatch, backend):
    host = separate_clocks(monkeypatch, step=.0001)
    if backend == "pylon":
        fake_pylon(monkeypatch)
        source = PylonSource(serial="00017")
    else:
        class Capture:
            def isOpened(self): return True
            def set(self, *_): return True
            def get(self, prop): return {cv2.CAP_PROP_FPS: 30, cv2.CAP_PROP_FRAME_WIDTH: 16,
                                       cv2.CAP_PROP_FRAME_HEIGHT: 12}.get(prop, 0)
            def getBackendName(self): return "clock-fixture"
            def read(self): return True, np.zeros((12, 16, 3), np.uint8)
            def release(self): pass
        monkeypatch.setattr(cv2, "VideoCapture", lambda *_: Capture())
        source = OpenCVSource(0)
    with source:
        frames = [source.read() for _ in range(8)]
        assert source.capabilities()["host_clock_domain"] == HOST_CLOCK_DOMAIN
        if backend == "pylon":
            assert source.capabilities()["serial"] == "00017"
            assert all(f.metadata["hardware_ticks"] == 987654321 for f in frames)
        stamps = [f.timestamp_s for f in frames]
        assert all(b > a for a, b in zip(stamps, stamps[1:]))
        assert stamps[-1] - stamps[0] < .0156
        assert all(100_000 <= f.timestamp_s <= f.received_monotonic_s <= host.value for f in frames)
        assert all(f.metadata["host_clock_domain"] == HOST_CLOCK_DOMAIN for f in frames)


def test_runtime_age_uses_receive_clock_and_preserves_independent_source_pts(monkeypatch):
    host = separate_clocks(monkeypatch)
    delivered, release = Event(), Event()

    class Source(FrameSource):
        def __init__(self): self.reads = 0
        def read(self):
            self.reads += 1
            if self.reads > 1:
                release.wait(2)
                return None
            return Frame(np.zeros((12, 16, 3), np.uint8), 2.5,
                         received_monotonic_s=host_now_s(),
                         metadata={"host_clock_domain": HOST_CLOCK_DOMAIN})
        def request_stop(self): release.set()
        def close(self): pass

    class Engine:
        def process(self, image, stamp):
            return GazeResult(stamp, True, "fixture", "clock-test", .9,
                              gaze_screen_px=(10, 20), screen_valid=True)
        def close(self): pass

    run = TrackingRuntime(Source, Engine, max_age_s=.2,
                          on_result=lambda *_: delivered.set())
    try:
        run.start()
        assert delivered.wait(2), "Runtime rejected a correctly timed source"
        fresh = run.snapshot(False)
        assert fresh["latest"]["valid"] and fresh["latest"]["timestamp_s"] == 2.5
        assert fresh["metrics"]["capture_age_ms"] == 0
        host.value += .25
        stale = run.snapshot(False)
        assert not stale["latest"]["valid"]
        assert stale["latest"]["reason"] == "stale_capture"
        assert stale["latest"]["gaze_screen_px"] is None
        assert stale["metrics"]["capture_age_ms"] == pytest.approx(250)
    finally:
        release.set()
        assert run.stop(1)

    def __init__(self, observation):

def face_landmarks(blink=False):
    intr = approximate_intrinsics(640, 480)
    xy, _ = cv2.projectPoints(FACE_TEMPLATE_M, np.zeros(3), np.array([0., 0., .65]),
                              intr.camera_matrix, intr.distortion)
    landmarks = {index: tuple(point) for index, point in zip(FACE_INDICES, xy.reshape(-1, 2))}
    # Preserve PnP outer corners and place realistic inner corners between them.
    landmarks[133] = (landmarks[33][0]+25, landmarks[33][1])
    landmarks[362] = (landmarks[263][0]-25, landmarks[263][1])
    aperture = .5 if blink else 6
    for iris, left, right, upper, lower, contour in (
        (468, 33, 133, 159, 145, (33, 160, 158, 133, 153, 144)),
        (473, 362, 263, 386, 374, (362, 385, 387, 263, 373, 380)),
    ):
        a, b = np.array(landmarks[left]), np.array(landmarks[right])
        midpoint = (a+b)/2
        landmarks[iris] = tuple(midpoint)
        landmarks[upper] = tuple(midpoint+[0, -aperture])
        landmarks[lower] = tuple(midpoint+[0, aperture])
        for j, fraction in ((1,.33), (2,.67), (4,.67), (5,.33)):
            landmarks[contour[j]] = tuple(a*(1-fraction)+b*fraction + [0, -aperture if j < 3 else aperture])
    return landmarks

def pump(app,condition,timeout=10):
    started=time.monotonic()
    while time.monotonic()-started<timeout:
        app.processEvents()
        if condition():return
        time.sleep(.01)
    raise AssertionError('Native application did not reach expected state')

def pump_until(app, condition, timeout=15):
    deadline = time.monotonic() + timeout
    while time.monotonic() < deadline:
        app.processEvents()
        if condition():
            return
        # QTest.qWait can retain Python's GIL around this manual event-pump
        # harness and starve QRunnable's Python callback. The production Qt
        # exec loop releases it; explicitly yield here to reproduce that.
        time.sleep(.01)
    raise AssertionError("Desktop worker did not reach the required state in time")

    finally:

        app.processEvents()
        window.pool.waitForDone(5000)

def test_replay_delayed_first_frame_seek_play_and_initial_gap_are_consistent(
        reviewed_window, monkeypatch):
    """Exercise native replay deterministically, independent of runner speed."""
    import numpy as np
    import gazespace.ui.main_window as desktop

    app, window, errors=reviewed_window
    window.timer.stop()  # drive the real poll handler with an explicit clock
    now=[100.0]
    with SessionRecorder(window.workspace/'sessions',
                         {'provenance':'synthetic','synthetic':True,'screen_size':[101,101]},
                         record_video=True,clock=lambda:now[0]) as recorder:
        for index,elapsed in enumerate((3.0,3.1004,3.2004)):
            now[0]=100.+elapsed
            recorder.append({'timestamp_s':10.+index/10,'valid':True,
                             'screen_valid':True,'spatial_valid':False,'quality':.9,
                             'reason':'synthetic_fixture','source':'synthetic',
                             'gaze_screen_px':[25.+index,50.],
                             'gaze_screen_normalized':[(25.+index)/100.,.5]},
                            np.full((120,160,3),index*50,np.uint8))
    window.load_session_path(recorder.path)
    assert window.replay.duration_s==pytest.approx(3.2004)
    assert window.replay_slider.maximum()==3201
    assert window.replay.current() is None
    assert window.replay_map.position is None
    assert window.replay_video._image is None

    # The rounded seek begins before the middle observation. One deterministic
    # timer tick must cross its actual recording-relative time, without a sleep.
    middle=window.current_session.records[1]
    window.replay_slider.setValue(round(middle['session_time_s']*1000))
    assert window._replay_index==0
    monkeypatch.setattr(desktop,'host_now_s',lambda:now[0])
    window.play_replay()
    now[0]+=.01
    window.poll()
    assert window.replay.position_s>=middle['session_time_s']
    assert window._replay_index==1
    assert window.replay_video._image is not None
    assert window.replay_map.position is not None
    window.pause_replay()

    # Scrubbing backwards must remove the future frame, overlay and trail, then
    # revisiting that same frame must reload it instead of using an old index.
    window.replay_slider.setValue(1000)
    assert window.replay.current() is None
    assert window._replay_index==-1
    assert window.replay_video._image is None
    assert window.replay_video._bgr is None
    assert not window.replay_video.result
    assert window.replay_video.observation is None
    assert window.replay_map.position is None
    assert window.replay_map.trail==[]
    window.replay_slider.setValue(3101)
    assert window._replay_index==1
    assert window.replay_video._image is not None
    assert window.replay_map.position is not None

    # The final slider step must reach a sub-millisecond final observation.
    window.replay_slider.setValue(window.replay_slider.maximum())
    assert window.replay.finished
    assert window._replay_index==2
    window.play_replay()
    assert window.replay.position_s==0
    assert window.replay.current() is None
    assert window.replay_video._image is None
    window.pause_replay()
    assert window._replay_clock is None
    assert not errors,errors

def pump(app, predicate, timeout=10):
    deadline = time.monotonic()+timeout
    while time.monotonic()<deadline:

class Clock:

    assert replay.current() is None  # no observation during initial delivery gap

    assert replay.seek(.21)['timestamp_s']==10.1
    assert replay.seek(.31)['timestamp_s']==.1
    assert replay.current()['replay'] is True

def fixture(kind='screen', *, synthetic=False):
    context = {'subject_id': 'person-A', 'camera_id': '0',
               'backend': 'synthetic' if synthetic else 'dshow'}
    rows = []
    if kind == 'screen':
        targets = [(x, y) for y in (.2, .5, .8) for x in (.1, .5, .9)]
        context['screen_size'] = [1920, 1080]
    else:
        targets = [(x, y, z) for z in (1., 1.3) for y in (-.15, .15) for x in (-.2, 0., .2)]
    for index, target in enumerate(targets):
        for repeat in range(4):
            x, y = target[:2]
            row = {'features': [.3+.4*x, .3+.2*y, .3+.4*x, .3+.2*y, 0, 0, 0, 0, 0, 0],
                   'target_id': str(index), 'session_id': 'training-session',
                   'time_s': 10+index+repeat*.04, 'quality': .9,
                   'context': deepcopy(context),
                   'acquisition': {'settle_seconds': .6, 'stable_duration_s': .12,
                                   'target_assumption': 'user_instructed_fixation'}}
            if kind == 'screen':
                row['target'] = list(target)
            else:
                origin = np.array([0., 0., .6])
                direction = np.asarray(target)-origin
                row.update(target=(direction/np.linalg.norm(direction)).tolist(),
                           target_world_m=list(target), eye_origin_camera_m=origin.tolist())
            rows.append(row)
    model = fit_gaze_model(rows, kind=kind, context=context, session_id='training-session',
                           screen_size=(1920, 1080) if kind == 'screen' else None,
                           do_group_validation=False)
    # This supplies expected static feature version/head conventions in these
    # mathematical fixtures; production callers use their observed tracker context.
    current = {**model.metadata, **context}
    fresh = deepcopy(rows)
    for index, row in enumerate(fresh):
        row['session_id'] = 'fresh-session'
        row['time_s'] += 100
        # Different acquired feature measurements, not merely renamed labels.
        row['features'][0] += (index % 4 + 1) * .0002
        row['features'][2] += (index % 4 + 1) * .0002
    return model, rows, fresh, current

class Source:
    def __init__(self,source='synthetic'):
        self.source=source
        self.closed=False
    def capabilities(self):
        return {'source':self.source,'backend':'synthetic','width':640,'height':480}
    def close(self): self.closed=True


class Tracker:
    def __init__(self):
        self.real=GazeTracker(mode='ir-roi',roi_normalized={'right':(.1,.2,.2,.2),'left':(.6,.2,.2,.2)})
        self.target=(.5,.5)
        self.closed=False
        self.blink=False
    def context(self,image_size=None):return self.real.context(image_size)
    def process(self,frame,t):
        x,y=self.target
        return Observation(t,(640,480),features=np.array([x,y,x,y,0,0,0,.5,.5,0]),
                           quality=.9,blink=self.blink,eyes_open=not self.blink,
                           pupil_method='ir_manual_roi',context=self.context((640,480)))
    def close(self): self.closed=True


def complete_session(session,tracker,t=0.):
    image=np.full((480,640,3),127,np.uint8)
    for _,point in session.targets:
        tracker.target=tuple(np.asarray(point)/(np.asarray(session.size)-1))
        session.start_target()
        first_t=t
        assert session.collecting
        for _ in range(100):
            t+=1/30
            progress=session.consume_frame(Frame(image,t))
            if not session.collecting:break
        assert progress['state']=='complete'
        assert t-first_t>=1.1
    assert session.completed
    return t

    assert not tracker.closed and not source.closed
    validate=ScreenSession(source,tracker,str(path),samples_per_target=4,validate=True,camera_id='synthetic')

def test_depth_coverage_checked_in_camera_frame_and_nonplanar_volume():
    # World Z has ample depth, but the camera looks along world Y: the actual
    # depth spread below is only 1 mm and is rejected despite world-Z variation.
    R=np.array([[1,0,0],[0,0,-1],[0,1,0]],float)
    ext=CameraExtrinsics(R,np.array([0,0,1.]))
    targets=[{'id':f'{x}-{y}-{z}','xyz_m':[x*.2,y*.001,z*.2]}
             for x in range(3) for y in range(2) for z in range(2)]
    with pytest.raises(ValueError,match='CAMERA'):_check_spatial_geometry(targets,ext,True)
    for target in targets:target['xyz_m'][1]*=100
    report=_check_spatial_geometry(targets,ext,True)
    assert report['camera_depth_span_m']==pytest.approx(.1)
    assert report['volumetric_target_coverage'] is True
    planar=[{'id':f'{x}-{y}','xyz_m':[x*.2,y*.2,0]} for x in range(3) for y in range(3)]
    with pytest.raises(ValueError,match='nonplanar'):_check_spatial_geometry(planar,ext,True)
    assert _check_spatial_geometry(planar,ext,False)['volumetric_target_coverage'] is False

def test_spatial_cli_workflow_collects_complete_targets_then_validates(tmp_path,monkeypatch):
    from gazespace.workflows import spatial_session
    from gazespace.types import HeadPose,CameraIntrinsics
    from gazespace.calibration import new_profile,save_json
    targets=[{'id':f'{x}-{y}-{z}','xyz_m':[x,y,z]}
             for x in (.1,.3,.5) for y in (.1,.4) for z in (0.,.2)]
    target_path=tmp_path/'targets.json'
    target_path.write_text(json.dumps(targets))
    profile_path=tmp_path/'spatial.json'
    intr=CameraIntrinsics(640,480,np.array([[640.,0,320],[0,640,240],[0,0,1]]),np.zeros((5,1)))
    ext=CameraExtrinsics(np.eye(3),np.array([0.,0.,1.]))
    save_json(profile_path,new_profile(intr,ext,camera_id='synthetic'))
    class SpatialTracker(Tracker):
        def process(self,frame,t):
            ob=super().process(frame,t)
            ob.head_pose=HeadPose(np.eye(3),np.array([.01,-.02,.6]),0,0,0,.1)
            ob.eye_origin_camera_m=np.array([.01,-.02,.6])
            ob.eye_origin_source='generic_face_pnp'
            return ob
    tracker=SpatialTracker()
    class SpatialSource(Source):
        def __init__(self):
            super().__init__()
            self.stamp=0.
            self.image=np.full((480,640,3),120,np.uint8)
        def read(self):
            self.stamp+=1/30
            return Frame(self.image,self.stamp)
    source=SpatialSource()
    pointer=[0]
    def ready(_):
        target=targets[pointer[0] % len(targets)]['xyz_m']
        tracker.target=(.3+target[0],.3+target[1])
        pointer[0]+=1
        return ''
    monkeypatch.setattr('builtins.input',ready)
    report=spatial_session(source,tracker,str(profile_path),str(target_path),samples_per_target=4)
    assert report['target_count']==12
    assert report['target_geometry']['volumetric_target_coverage'] is True
    saved=load_profile(profile_path)
    assert saved['spatial_model']['metadata']['eye_origin_source']=='generic_face_pnp'
    # The deterministic tracker reproduces exactly the original feature/target
    # arrays for these targets. New timestamps/session labels cannot turn that
    # fixture into independent evidence.
    with pytest.raises(ValueError,match='exact model training observations'):
        spatial_session(source,tracker,str(profile_path),str(target_path),samples_per_target=4,validate=True)
    # Acquire a genuinely different held-out target set for the successful path.
    for target in targets:
        target['xyz_m']=(np.asarray(target['xyz_m'])+.01).tolist()
    target_path.write_text(json.dumps(targets))
    validation=spatial_session(source,tracker,str(profile_path),str(target_path),samples_per_target=4,validate=True)
    assert validation['independent_session'] is True
    assert validation['angular_error']['count']>=12*4
    assert load_profile(profile_path)['spatial_validation']['session_id']==validation['session_id']
    assert not source.closed and not tracker.closed
